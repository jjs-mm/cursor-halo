cmake_minimum_required(VERSION 3.16)
project(CursorHalo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" CURSOR_HALO_VERSION LIMIT_COUNT 1)
if(NOT CURSOR_HALO_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "Invalid release VERSION")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(DESKTOP REQUIRED IMPORTED_TARGET gtk+-3.0 gio-unix-2.0 x11 xi xfixes xtst xext)
add_executable(cursor-halo src/main.cpp)
target_link_libraries(cursor-halo PRIVATE PkgConfig::DESKTOP)
target_compile_definitions(cursor-halo PRIVATE CURSOR_HALO_VERSION="${CURSOR_HALO_VERSION}")
target_compile_options(cursor-halo PRIVATE -Wall -Wextra -Wpedantic)
add_executable(cursor-halo-core-tests tests/core_test.cpp)
target_compile_options(cursor-halo-core-tests PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
enable_testing()
add_test(NAME core COMMAND cursor-halo-core-tests)
